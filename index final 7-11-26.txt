                <div class="footer-brand" style="margin-top: 15px;">
                    Sounds provided by <a href="https://www.myinstants.com" target="_blank">MyInstants.com</a><br>
                    Optimizado para streaming profesional.
                </div>
            </div>

        </div>
    </div>

    <script src="/socket.io/socket.io.js"></script>
    <script>
        const socket = io({
            reconnection: true,
            reconnectionAttempts: Infinity,
            reconnectionDelay: 1000,
            reconnectionDelayMax: 5000,
            timeout: 20000
        });

        document.addEventListener("visibilitychange", () => {
            if (document.visibilityState === "visible") {
                if (!socket.connected) {
                    socket.connect();
                }
            }
        });

        document.addEventListener('keydown', (event) => {
            if (event.target.tagName === 'INPUT' || event.target.tagName === 'SELECT' || event.target.tagName === 'TEXTAREA') {
                return;
            }

            if (event.key >= '1' && event.key <= '9') {
                reproducirEfectoDirecto(`sound${event.key}`);
            } else if (event.key === '0') {
                reproducirEfectoDirecto('sound10');
            }
        });

        let currentLicenseKey = '';
        let colaAudios = [];
        let estaReproduciendoAudio = false;
        let volumenGlobal = 1.0;
        let ttsSilenciado = false;
        let estaConectadoBot = false;

        function togglePasswordVisibility() {
            const inputKey = document.getElementById('licenseKeyInput');
            const btnToggle = document.querySelector('.toggle-password');
            if (inputKey.type === 'password') {
                inputKey.type = 'text';
                btnToggle.innerText = '🙈';
            } else {
                inputKey.type = 'password';
                btnToggle.innerText = '👁️';
            }
        }

        function validarLicencia() {
            const key = document.getElementById('licenseKeyInput').value.trim();
            const errorDiv = document.getElementById('loginError');
            if (!key) {
                errorDiv.innerText = "Por favor ingresa una clave de acceso.";
                return;
            }
            errorDiv.innerText = "Validando licencia...";
            socket.emit('validar-licencia', { key: key });
        }

        socket.on('respuesta-licencia', (data) => {
            if (data.valida) {
                currentLicenseKey = data.key;
                document.getElementById('loginOverlay').style.display = 'none';
                document.getElementById('mainApp').style.display = 'flex';
                
                if (data.expiracion) {
                    document.getElementById('infoLicencia').innerText = `🔑 ${data.expiracion}`;
                }
            } else {
                document.getElementById('loginError').innerText = data.mensaje || "Clave de acceso inválida o vencida.";
            }
        });

        socket.on('sesion-expulsada', (data) => {
            alert(data.mensaje || "Tu clave de acceso fue abierta en otro dispositivo.");
            location.reload();
        });

        function actualizarVolumen(valor) {
            document.getElementById('volValue').innerText = valor;
            volumenGlobal = valor / 100;
        }

        function toggleMuteTTS() {
            ttsSilenciado = !ttsSilenciado;
            const btn = document.getElementById('btnMuteTTS');
            if (ttsSilenciado) {
                btn.innerText = "🔇 Muteado";
                btn.style.background = "#ef4444";
                colaAudios = [];
            } else {
                btn.innerText = "🔊 Voz Activa";
                btn.style.background = "#334155";
            }
        }

        function obtenerArchivoBase64(inputId) {
            return new Promise((resolve) => {
                const fileInput = document.getElementById(inputId);
                if (fileInput.files && fileInput.files[0]) {
                    const reader = new FileReader();
                    reader.onload = function(e) { resolve(e.target.result); };
                    reader.readAsDataURL(fileInput.files[0]);
                } else {
                    resolve(null);
                }
            });
        }

        async function conectarBot() {
            const btnConectar = document.getElementById('btnConectarBot');

            if (estaConectadoBot) {
                socket.emit('desconectar-bot');
                resetearEstadoBot();
                return;
            }

            let user = document.getElementById('username').value.trim();
            if (!user) {
                alert("Por favor ingresa un usuario válido");
                return;
            }

            if (!user.startsWith('@')) user = '@' + user;

            let idiomaVoz = document.getElementById('idiomaVoz').value;
            let leerNombre = document.getElementById('leerNombre').checked;
            let leerEmojis = document.getElementById('leerEmojis').checked;
            
            let roles = {
                todos: document.getElementById('rolTodos').checked,
                seguidores: document.getElementById('rolSeguidores').checked,
                moderadores: document.getElementById('rolModeradores').checked,
                suscriptores: document.getElementById('rolSuscriptores').checked
            };

            let tipoComentario = document.querySelector('input[name="tipoComentario"]:checked').value;
            let textoComando = document.getElementById('textoComando').value.trim();

            let inputProhibidas = document.getElementById('palabrasProhibidas').value.trim();
            let listaProhibidas = inputProhibidas 
                ? inputProhibidas.split(',').map(p => p.trim()).filter(p => p.length > 0)
                : [];

            let soundFollowBase64 = await obtenerArchivoBase64('soundFollow');
            let soundGiftBase64 = await obtenerArchivoBase64('soundGift');

            let alertas = {
                follow: document.getElementById('alertaFollow').checked,
                gift: document.getElementById('alertaGift').checked,
                soundFollow: soundFollowBase64,
                soundGift: soundGiftBase64
            };

            document.getElementById('status').innerText = "Estado: Conectando... 🟡";
            
            socket.emit('configurar-usuario', { 
                licenseKey: currentLicenseKey,
                username: user, 
                idiomaVoz: idiomaVoz,
                leerNombre: leerNombre,
                leerEmojis: leerEmojis,
                roles: roles,
                tipoComentario: tipoComentario,
                textoComando: textoComando,
                palabrasProhibidas: listaProhibidas,
                alertas: alertas
            });
        }

        function resetearEstadoBot() {
            estaConectadoBot = false;
            const btn = document.getElementById('btnConectarBot');
            btn.innerText = "Conectar Bot";
            btn.style.background = "#fe2c55";
            document.getElementById('status').innerText = "Estado: Desconectado 🔴";
        }

        async function probarAlerta(tipo) {
            let soundBase64 = await obtenerArchivoBase64(tipo === 'follow' ? 'soundFollow' : 'soundGift');
            let idiomaVoz = document.getElementById('idiomaVoz').value;
            socket.emit('probar-alerta', { tipo: tipo, customSound: soundBase64, idioma: idiomaVoz });
        }

        function probarEfectoDirecto(idInput) {
            document.getElementById(idInput).click();
        }

        document.addEventListener('DOMContentLoaded', () => {
            for (let i = 1; i <= 10; i++) {
                const savedAudio = localStorage.getItem(`soundboard_audio_${i}`);
                const savedName = localStorage.getItem(`soundboard_name_${i}`);

                if (savedAudio && savedName) {
                    window[`audioBase64_${i}`] = savedAudio;
                    const teclaLabel = i === 10 ? '0' : i;
                    const btnPlay = document.getElementById(`btnPlay${i}`);
                    if (btnPlay) {
                        btnPlay.innerText = `🔊 [${teclaLabel}] ${savedName}`;
                    }
                }
            }

            for (let i = 1; i <= 10; i++) {
                const input = document.getElementById(`sound${i}`);
                if (input) {
                    input.addEventListener('change', function() {
                        if (this.files && this.files[0]) {
                            const file = this.files[0];
                            const nombreOriginal = file.name;
                            const nombreCorto = nombreOriginal.length > 10 ? nombreOriginal.substring(0, 8) + '...' : nombreOriginal;
                            
                            const reader = new FileReader();
                            reader.onload = function(e) {
                                const base64Result = e.target.result;
                                localStorage.setItem(`soundboard_audio_${i}`, base64Result);
                                localStorage.setItem(`soundboard_name_${i}`, nombreCorto);
                                window[`audioBase64_${i}`] = base64Result;
                                
                                const teclaLabel = i === 10 ? '0' : i;
                                const btnPlay = document.getElementById(`btnPlay${i}`);
                                if (btnPlay) {
                                    btnPlay.innerText = `🔊 [${teclaLabel}] ${nombreCorto}`;
                                }
                            };
                            reader.readAsDataURL(file);
                        }
                    });
                }
            }
        });

        async function reproducirEfectoDirecto(idInput) {
            const canalNum = idInput.replace('sound', '');
            let soundBase64 = window[`audioBase64_${canalNum}`] || await obtenerArchivoBase64(idInput);

            if (!soundBase64) {
                alert("Primero selecciona un archivo de audio para este canal con el botón [⚙️].");
                return;
            }
            socket.emit('reproducir-efecto-rapido', { customSound: soundBase64 });
        }

        socket.on('estado-conexion', (data) => {
            if (data.status === 'conectado') {
                estaConectadoBot = true;
                const btn = document.getElementById('btnConectarBot');
                btn.innerText = "Desconectar Bot";
                btn.style.background = "#334155";
                document.getElementById('status').innerText = "Estado: Conectado 🟢";
                document.getElementById('chatFeed').innerHTML = '<div class="chat-msg" style="color: #38bdf8; border-left-color: #38bdf8;">Conectado correctamente al chat y alertas.</div>';
            } else if (data.status === 'desconectado') {
                resetearEstadoBot();
            } else {
                resetearEstadoBot();
                document.getElementById('status').innerText = "Estado: Error de conexión ❌";
            }
        });

        // Actualizar el Ranking de Actividad
        socket.on('actualizar-ranking-likes', (topLikes) => {
            const container = document.getElementById('listaRankingPanel');
            if (!container) return;

            if (!topLikes || topLikes.length === 0) {
                container.innerHTML = `<div style="text-align: center; color: #64748b; padding: 15px 0;">Esperando actividad...</div>`;
                return;
            }

            container.innerHTML = topLikes.map((item, index) => `
                <div style="display: flex; justify-content: space-between; align-items: center; padding: 5px 0; border-bottom: 1px solid #1e293b;">
                    <span><strong style="color: #38bdf8;">#${index + 1}</strong> ${item.user}</span>
                    <span style="color: #4ade80; font-weight: bold;">${item.likes.toLocaleString()}</span>
                </div>
            `).join('');
        });

        function reiniciarRanking() {
            fetch('/api/reset-likes', { method: 'POST' })
                .then(res => res.json())
                .then(() => console.log('Ranking reiniciado'));
        }

        socket.off('reproducir-audio');
        socket.on('reproducir-audio', (data) => {
            if (!data) return;

            if (!ttsSilenciado) {
                if (data.soundUrl) {
                    colaAudios.push({ url: data.soundUrl, isCustomSound: true });
                }

                if (data.audioUrl) {
                    colaAudios.push({ url: data.audioUrl, isCustomSound: false });
                }
            }

            const chatFeed = document.getElementById('chatFeed');
            
            if (chatFeed.children.length === 1 && chatFeed.children[0].innerText.includes('Esperando')) {
                chatFeed.innerHTML = '';
            }

            if (data.mensaje) {
                const divMsg = document.createElement('div');
                divMsg.className = data.esAlerta ? 'chat-msg alert-msg' : 'chat-msg';
                divMsg.innerHTML = `<strong>${data.usuario || 'Sistema'}:</strong> ${data.mensaje}`;
                chatFeed.appendChild(divMsg);
                chatFeed.scrollTop = chatFeed.scrollHeight;
            }

            if (!ttsSilenciado) {
                procesarColaAudios();
            }
        });

        function procesarColaAudios() {
            if (ttsSilenciado || estaReproduciendoAudio || colaAudios.length === 0) return;

            estaReproduciendoAudio = true;
            const itemAudio = colaAudios.shift();

            const audio = new Audio(itemAudio.url);
            audio.volume = volumenGlobal;

            audio.play().catch(err => {
                console.log("Error al reproducir audio:", err);
                siguienteAudio();
            });

            audio.onended = () => {
                siguienteAudio();
            };
        }

        function siguienteAudio() {
            estaReproduciendoAudio = false;
            setTimeout(procesarColaAudios, 100);
        }
    </script>
